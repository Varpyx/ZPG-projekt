Shader: // Compiles one shader of the given type and returns its id
Shader: static GLuint createShaderFromFile(GLenum type, const char* shaderFile)
Shader: {
Shader: 	// Creates an empty shader
Shader: 	GLuint shaderID = glCreateShader(type);
Shader: 
Shader: 	if (shaderID == 0)
Shader: 	{
Shader: 		std::cout << "Unable to create shader" << std::endl;
Shader: 		exit(EXIT_FAILURE);
Shader: 	}
Shader: 
Shader: 	//Loading the contents of a file into a variable
Shader: 	std::ifstream file(shaderFile);
Shader: 	if (!file.is_open())
Shader: 	{
Shader: 		std::cout << "Unable to open file " << shaderFile << std::endl;
Shader: 		glDeleteShader(shaderID);
Shader: 		exit(-1);
Shader: 	}
Shader: 	std::string shaderCode((std::istreambuf_iterator<char>(file)),
Shader: 		std::istreambuf_iterator<char>());
Shader: 
Shader: 	// Set the shader source code
Shader: 	const char* source = shaderCode.c_str();
Shader: 	glShaderSource(shaderID, 1, &source, nullptr);
Shader: 
Shader: 	// Compile the shader source code
Shader: 	glCompileShader(shaderID);
Shader: 
Shader: 	// Check specialization/compilation status
Shader: 	int success;
Shader: 	glGetShaderiv(shaderID, GL_COMPILE_STATUS, &success);
Shader: 	if (!success)
Shader: 	{
Shader: 		char infoLog[1024];
Shader: 		glGetShaderInfoLog(shaderID, sizeof(infoLog), nullptr, infoLog);
Shader: 		std::cout
Shader: 			<< "Shader failed:\n"
Shader: 			<< infoLog << std::endl;
Shader: 		glDeleteShader(shaderID);
Shader: 		exit(1);
Shader: 	}
Shader: 	return shaderID;
Shader: }
Shader:
